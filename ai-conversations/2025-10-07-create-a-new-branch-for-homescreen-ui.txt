    // Layer 2: Domain Cache Service - Persistent storage
    class CacheService implements ICacheService {
      constructor(private storage: IStorageService) {}
      
      async getCachedMedia(id: string): Promise<Media | null> {
        const cached = await this.storage.get(`media:${id}`)
        if (cached && !this.isExpired(cached)) {
          return cached.data
        }
        return null
      }
      
      // Intelligent cache warming
      async warmCache(mediaIds: string[]): Promise<void> {
        const promises = mediaIds.map(id => this.preloadMedia(id))
        await Promise.allSettled(promises)
      }
    }

    // Layer 3: API-side caching hints
    class OptimizedHttpClient {
      // Smart request batching
      async batchMediaRequests(mediaIds: string[]): Promise<Media[]> {
        const cached = await this.getCachedBatch(mediaIds)
        const missing = mediaIds.filter(id => !cached.has(id))
        
        if (missing.length > 0) {
          const fresh = await this.fetchBatch(missing)
          return [...cached.values(), ...fresh]
        }
        
        return [...cached.values()]
      }
      
      // Cache-aware API calls
      async get<T>(endpoint: string, cacheHints?: CacheHints): Promise<T> {
        const headers = this.buildCacheHeaders(cacheHints)
        return this.client.get(endpoint, { headers })
      }
    }
